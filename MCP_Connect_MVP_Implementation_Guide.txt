MCP Connect MVP
Implementation Guide for Claude Code
Simple Cross-Platform MCP Client
What This Builds:
A desktop/mobile app that connects to MCP servers, discovers their tools, lets users invoke them via a simple UI, and uses OpenAI&apos;s API to provide natural language interaction.
Tech Stack:
• Tauri 2.0 (Rust + TypeScript)
• React 18 + TypeScript
• TailwindCSS for styling
• MCP TypeScript SDK (@modelcontextprotocol/sdk)
• OpenAI API (gpt-4o)
Build Time: ~4-6 hours for experienced developer

4.2 MCP Client Service
File: src/services/mcpClient.ts
import { MCPServer, MCPTool } from &apos;../types/mcp&apos;;
export class MCPClient {
 private servers: Map&lt;string, MCPServer&gt; = new Map();
 private tools: Map&lt;string, MCPTool[]&gt; = new Map();
 async connectServer(server: MCPServer): Promise&lt;void&gt; {
 try {
 // Update status
 server.status = &apos;connecting&apos;;
 this.servers.set(server.id, server);
 // HTTP: POST to /initialize
 if (server.transport === &apos;http&apos;) {
 const response = await fetch(`${server.url}/initialize`, {
 method: &apos;POST&apos;,
 headers: { &apos;Content-Type&apos;: &apos;application/json&apos; },
 body: JSON.stringify({
 protocolVersion: &apos;2024-11-05&apos;,
 capabilities: { tools: {} },
 clientInfo: { name: &apos;mcp-connect-mvp&apos;, version: &apos;1.0.0&apos; }
 })
 });
 if (!response.ok) throw new Error(&apos;Failed to initialize&apos;);
 
 // List tools
 const toolsResp = await fetch(`${server.url}/tools/list`, {
 method: &apos;POST&apos;,
 headers: { &apos;Content-Type&apos;: &apos;application/json&apos; },
 body: JSON.stringify({})
 });
 const toolsData = await toolsResp.json();
 this.tools.set(server.id, toolsData.tools || []);
 }
 server.status = &apos;connected&apos;;
 this.servers.set(server.id, server);
 } catch (error) {
 server.status = &apos;error&apos;;
 this.servers.set(server.id, server);
 throw error;
 }
 }
 async callTool(
 serverId: string,
 toolName: string,
 args: Record&lt;string, any&gt;
 ): Promise&lt;any&gt; {
 const server = this.servers.get(serverId);
 if (!server) throw new Error(&apos;Server not found&apos;);
 if (server.status !== &apos;connected&apos;) throw new Error(&apos;Server not connected&apos;);
 const response = await fetch(`${server.url}/tools/call`, {
 method: &apos;POST&apos;,
 headers: { &apos;Content-Type&apos;: &apos;application/json&apos; },
 body: JSON.stringify({
 name: toolName,
 arguments: args
 })
 });
 if (!response.ok) {
 const error = await response.json();
 throw new Error(error.error?.message || &apos;Tool call failed&apos;);
 }
 const result = await response.json();
 return result.content;
 }
 getTools(serverId: string): MCPTool[] {
 return this.tools.get(serverId) || [];
 }
 getAllServers(): MCPServer[] {
 return Array.from(this.servers.values());
 }
}
export const mcpClient = new MCPClient();
4.3 OpenAI Client Service
File: src/services/openaiClient.ts
import OpenAI from &apos;openai&apos;;
import { MCPTool, Message, ToolCall } from &apos;../types/mcp&apos;;
import { mcpClient } from &apos;./mcpClient&apos;;
export class OpenAIClient {
 private client: OpenAI | null = null;
 private apiKey: string = &apos;&apos;;
 setApiKey(key: string) {
 this.apiKey = key;
 this.client = new OpenAI({
 apiKey: key,
 dangerouslyAllowBrowser: true
 });
 }
 async chat(
 messages: Message[],
 availableTools: Map&lt;string, { serverId: string; tools: MCPTool[] }&gt;
 ): Promise&lt;{ message: Message; toolCalls: ToolCall[] }&gt; {
 if (!this.client) throw new Error(&apos;OpenAI client not initialized&apos;);
 // Convert MCP tools to OpenAI format
 const tools: OpenAI.Chat.ChatCompletionTool[] = [];
 const toolMap = new Map&lt;string, { serverId: string; tool: MCPTool }&gt;();
 for (const [serverId, { tools: serverTools }] of availableTools) {
 for (const tool of serverTools) {
 tools.push({
 type: &apos;function&apos;,
 function: {
 name: `${serverId}__${tool.name}`,
 description: tool.description,
 parameters: tool.inputSchema
 }
 });
 toolMap.set(`${serverId}__${tool.name}`, { serverId, tool });
 }
 }
 // Call OpenAI
 const completion = await this.client.chat.completions.create({
 model: &apos;gpt-4o&apos;,
 messages: messages.map(m =&gt; ({
 role: m.role,
 content: m.content
 })),
 tools: tools.length &gt; 0 ? tools : undefined,
 tool_choice: &apos;auto&apos;
 });
 const response = completion.choices[0].message;
 const toolCalls: ToolCall[] = [];
 // Execute tool calls
 if (response.tool_calls &amp;&amp; response.tool_calls.length &gt; 0) {
 for (const tc of response.tool_calls) {
 const toolInfo = toolMap.get(tc.function.name);
 if (!toolInfo) continue;
 const toolCall: ToolCall = {
 id: tc.id,
 serverId: toolInfo.serverId,
 toolName: toolInfo.tool.name,
 arguments: JSON.parse(tc.function.arguments),
 status: &apos;pending&apos;
 };
 try {
 toolCall.status = &apos;running&apos;;
 const result = await mcpClient.callTool(
 toolInfo.serverId,
 toolInfo.tool.name,
 toolCall.arguments
 );
 toolCall.result = result;
 toolCall.status = &apos;completed&apos;;
 } catch (error) {
 toolCall.error = error instanceof Error ? error.message : &apos;Unknown error&apos;;
 toolCall.status = &apos;failed&apos;;
 }
 toolCalls.push(toolCall);
 }
 }
 return {
 message: {
 id: crypto.randomUUID(),
 role: &apos;assistant&apos;,
 content: response.content || &apos;&apos;,
 timestamp: new Date()
 },
 toolCalls
 };
 }
}
export const openaiClient = new OpenAIClient();
